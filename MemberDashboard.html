async function createCampaign() { const name = document.getElementById("campaignName").value.trim(); const advertiser = document.getElementById("advertiserName").value.trim(); const type = document.getElementById("commissionType").value; const fixed = Number(document.getElementById("fixedAmount").value || 0); const percentage = Number(document.getElementById("percentageRate").value || 0); const button = document.getElementById("createCampaignButton"); if (!name || !advertiser) { showMessage("Please enter the campaign name and advertiser name."); return; } button.disabled = true; button.textContent = "Creating campaign..."; try { const campaignData = { name: name, campaign_name: name, advertiser_name: advertiser, partner_nane: advertiser, campaign_status: "active", commission_type: type, fixed_amount: fixed, percentage_rate: percentage }; const { data, error } = await db .from("campaigns") .insert([campaignData]) .select(); if (error) { console.error("Campaign creation error:", error); showMessage("CAMPAIGN ERROR: " + error.message + " | Code: " + error.code); return; } showMessage("Campaign created successfully!"); document.getElementById("campaignName").value = ""; document.getElementById("advertiserName").value = ""; document.getElementById("fixedAmount").value = ""; document.getElementById("percentageRate").value = ""; await loadAdmin(); } catch (error) { console.error(error); showMessage("ERROR: " + error.message); } finally { button.disabled = false; button.textContent = "Create Campaign"; } }
