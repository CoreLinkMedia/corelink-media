" | Code: " + error.code : "")); return; } showMessage("Campaign created successfully."); document.getElementById("campaignName").value = ""; document.getElementById("advertiserName").value = ""; document.getElementById("fixedAmount").value = ""; document.getElementById("percentageRate").value = ""; await loadAdmin(); }catch(error){ console.error("Campaign error:",error); showMessage("CAMPAIGN ERROR: " + error.message); }finally{ button.disabled = false; button.textContent = "Create Campaign"; } }
