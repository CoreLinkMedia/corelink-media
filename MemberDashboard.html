async function createCampaign() { const name = document.getElementById("campaignName").value.trim(); const advertiser = document.getElementById("advertiserName").value.trim(); const type = document.getElementById("commissionType").value; const fixed = Number(document.getElementById("fixedAmount").value || 0); const percentage = Number(document.getElementById("percentageRate").value || 0); const button = document.getElementById("createCampaignButton"); if (!name || !advertiser) { showMessage("Enter campaign name and advertiser name."); return; } button.disabled = true; button.textContent = "Creating..."; try { const { error } = await db.from("campaigns").insert({ name: name, partner_name: advertiser, campaign_name: name, advertiser_name: advertiser, commision_type: type, commision_value: type === "percentage" ? percentage : fixed, campaign_status: "active", status: "active", fixed_amount: fixed, percentage_rate: percentage }); if (error) { console.error("Campaign creation error:", error); showMessage("CAMPAIGN ERROR: " + error.message + " | Code: " + error.code); return; } showMessage("Campaign created successfully!"); document.getElementById("campaignName").value = ""; document.getElementById("advertiserName").value = ""; document.getElementById("fixedAmount").value = ""; document.getElementById("percentageRate").value = ""; await loadAdmin(); } catch (error) { showMessage("CAMPAIGN ERROR: " + error.message); } finally { button.disabled = false; button.textContent = "Create Campaign"; } }
